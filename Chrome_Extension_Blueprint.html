▼ 🧩 Step 2: Manifest V3 & World Isolation (MAIN vs ISOLATED)

Manifest V3 enforces script world isolation: interceptor.js (which monkey-patches window.fetch to catch network API payloads) MUST run in "world": "MAIN", but MAIN world scripts cannot access chrome.runtime or chrome.storage APIs. Communication must bridge via Custom DOM Events.

2.1 Manifest World Isolation Configuration

{
  "manifest_version": 3,
  "name": "TUFHub - TakeUForward to GitHub Sync",
  "version": "1.0.0",
  "permissions": ["storage", "declarativeNetRequest"],
  "host_permissions": ["*://takeuforward.org/*", "*://api.github.com/*"],
  "background": {
    "service_worker": "scripts/background.js",
    "type": "module"
  },
  "content_scripts": [
    {
      "matches": ["https://takeuforward.org/plus/*"],
      "js": ["scripts/tuf/interceptor.js"],
      "run_at": "document_start",
      "world": "MAIN"
    },
    {
      "matches": ["https://takeuforward.org/plus/*"],
      "js": ["scripts/tuf/content.js"],
      "run_at": "document_idle"
    }
  ]
}

2.2 Custom Event Communication Bridge

// interceptor.js (MAIN World - Intercepts raw fetch payload)
const origFetch = window.fetch;
window.fetch = async function(...args) {
  const res = await origFetch.apply(this, args);
  const clone = res.clone();
  try {
    const data = await clone.json();
    if (data && data.status === 'ACCEPTED' && data.code) {
      window.dispatchEvent(new CustomEvent('TUFHUB_ACCEPTED_SUBMISSION', {
        detail: { code: data.code, language: data.language, title: data.title }
      }));
    }
  } catch(e) {}
  return res;
};

// content.js (ISOLATED World - Receives event and talks to chrome.runtime)
window.addEventListener('TUFHUB_ACCEPTED_SUBMISSION', async (event) => {
  const { code, language, title } = event.detail;
  chrome.runtime.sendMessage({ type: 'SYNC_SUBMISSION', payload: { code, language, title } });
});

▼ 🔐 Step 3: Tab-Based OAuth Fallback & Auth Security

Standard chrome.identity.launchWebAuthFlow fails silently inside Brave Browser or Chrome instances with strict privacy extensions. To guarantee 100% auth success across all browsers, implement a tab URL listener fallback in background.js:

// background.js Tab Listener for Privacy-Shielded Browsers
chrome.tabs.onUpdated.addListener(async (tabId, changeInfo, tab) => {
  if (changeInfo.url && changeInfo.url.includes('.chromiumapp.org/?code=')) {
    const url = new URL(changeInfo.url);
    const code = url.searchParams.get('code');
    if (code) {
      const token = await exchangeCodeForToken(code);
      await chrome.storage.local.set({ token });
      chrome.tabs.remove(tabId);
      chrome.tabs.create({ url: 'welcome.html?status=success' });
    }
  }
});

▼ ⚡ Step 4: Git 409 Conflict Retry & Root README Indexer

Rapid commits to the GitHub REST API cause Git 409 Conflict status codes due to outdated file SHAs. uploader.js implements automatic SHA re-fetching and retry logic:

4.1 Git 409 Conflict Retry Handler (uploader.js)

async function uploadFileToGitHub(token, repo, filePath, content, commitMsg, sha = '') {
  const url = `https://api.github.com/repos/${repo}/contents/${filePath}`;
  const body = { message: commitMsg, content: btoa(unescape(encodeURIComponent(content))) };
  if (sha) body.sha = sha;

  let res = await fetch(url, {
    method: 'PUT',
    headers: { Authorization: `token ${token}`, Accept: 'application/vnd.github.v3+json' },
    body: JSON.stringify(body)
  });

  // Handle Git 409 Conflict: Re-fetch latest SHA and retry once
  if (res.status === 409) {
    const getRes = await fetch(url, {
      headers: { Authorization: `token ${token}`, Accept: 'application/vnd.github.v3+json' }
    });
    if (getRes.ok) {
      const getJson = await getRes.json();
      body.sha = getJson.sha;
      res = await fetch(url, {
        method: 'PUT',
        headers: { Authorization: `token ${token}`, Accept: 'application/vnd.github.v3+json' },
        body: JSON.stringify(body)
      });
    }
  }
  return res.ok;
}

▼ 🐍 Step 6: Python Pillow Asset Automation Codebase

Complete Pillow python automation script for generating glowing presentation cards:

import os
from PIL import Image, ImageDraw, ImageFilter

def create_glowing_store_card(src_path, out_path, title_text, url_text="https://takeuforward.org/plus/dsa"):
    W, H = 1280, 800
    bg = Image.new("RGBA", (W, H), (11, 17, 32, 255))
    
    # 1. Multi-layered Radial Glow Spheres
    glow = Image.new("RGBA", (W, H), (0, 0, 0, 0))
    gdraw = ImageDraw.Draw(glow)
    gdraw.ellipse((800, -100, 1400, 500), fill=(249, 115, 22, 90)) # Fire Orange
    gdraw.ellipse((-200, 400, 500, 1000), fill=(37, 99, 235, 90)) # Royal Blue
    glow = glow.filter(ImageFilter.GaussianBlur(120))
    bg = Image.alpha_composite(bg, glow)
    draw = ImageDraw.Draw(bg)
    
    # 2. Window Mockup & Header
    win_w, win_h = 1060, 620
    win_x = (W - win_w) // 2
    win_y = 110
    
    draw.rounded_rectangle((win_x, win_y, win_x+win_w, win_y+win_h), radius=12, fill=(30, 30, 38, 255), outline=(255,255,255,20))
    # Traffic Lights (🔴 🟡 🟢)
    draw.ellipse((win_x+16, win_y+16, win_x+28, win_y+28), fill=(255, 95, 86, 255))
    draw.ellipse((win_x+36, win_y+16, win_x+48, win_y+28), fill=(255, 189, 46, 255))
    draw.ellipse((win_x+56, win_y+16, win_x+68, win_y+28), fill=(39, 201, 63, 255))
    
    # Vector Status Dot (Green Circle)
    draw.ellipse((win_x+500, win_y+19, win_x+506, win_y+25), fill=(39, 201, 63, 255))
    
    if os.path.exists(src_path):
        img = Image.open(src_path).convert("RGBA")
        img = img.resize((win_w, win_h - 44), Image.Resampling.LANCZOS)
        bg.paste(img, (win_x, win_y + 44), img)
        
    bg.convert("RGB").save(out_path, quality=95)
    print(f"Generated Glowing Presentation Card: {out_path}")
